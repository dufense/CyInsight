42592→    try {
42593→      const { source, type, search, dateFrom, dateTo, limit = "100", offset = "0" } = req.query;
42594→      const lim = Math.min(parseInt(String(limit), 10) || 100, 500);
42595→      const off = parseInt(String(offset), 10) || 0;
42596→
42597→      const whereParts: string[] = [];
42598→      const params: unknown[] = [];
42599→
42600→      if (source && source !== "all") {
42601→        if (source === "opencti") {
42602→          params.push("opencti");
42603→          whereParts.push(`source = $${params.length}`);
42604→        } else if (source === "taxii") {
42605→          params.push("taxii:%");
42606→          whereParts.push(`source LIKE $${params.length}`);
42607→        } else {
42608→          params.push(String(source));
42609→          whereParts.push(`source = $${params.length}`);
42610→        }
42611→      }

42614→          whereParts.push(`indicator_type IN ('hash_md5'::ioc_type, 'hash_sha256'::ioc_type, 'hash_sha1'::ioc_type)`);
42615→        } else {
42616→          params.push(type);
42617→          whereParts.push(`indicator_type = $${params.length}::ioc_type`);
42618→        }
42619→      }
42620→      if (search) {
42621→        params.push(`%${search}%`);
42622→        whereParts.push(`indicator_value ILIKE $${params.length}`);
42623→      }
42624→      if (dateFrom) {
42625→        params.push(dateFrom);
42626→        whereParts.push(`first_seen >= $${params.length}`);
42627→      }
42628→      if (dateTo) {
42629→        params.push(dateTo);
42630→        whereParts.push(`first_seen <= $${params.length}`);
42631→      }
42632→
42633→      const where = whereParts.length > 0 ? `WHERE ${whereParts.join(" AND ")}` : "";
42634→
42635→      // Proper UNION ALL query for correct mixed-source pagination
42636→      const unionQuery = `
42637→        SELECT stix_id, indicator_type, indicator_value, reputation, confidence, source,
42638→               tags::text AS labels, first_seen, last_seen, updated_at, 'taxii' AS source_type
42639→        FROM taxii_stix_iocs ${where}

42641→        SELECT stix_id, indicator_type, indicator_value, reputation, confidence, source,
42642→               labels::text AS labels, first_seen, last_seen, updated_at, 'opencti' AS source_type
42643→        FROM opencti_ioc_cache ${where}
42644→      `;
42645→      const dataParams = [...params, lim, off];
42646→      const [dataResult, countResult] = await Promise.all([
42647→        pool.query(

42649→          dataParams
42650→        ),
42651→        pool.query(
42652→          `SELECT COUNT(*) FROM (${unionQuery}) u`,
42653→          params
42654→        ),
42655→      ]);

42658→
42659→      // Get source stats from both TAXII and OpenCTI caches
42660→      const sourcesResult = await pool.query(
42661→        `SELECT source, COUNT(*) AS count FROM taxii_stix_iocs GROUP BY source
42662→         UNION ALL
42663→         SELECT source, COUNT(*) AS count FROM opencti_ioc_cache GROUP BY source`
42664→      );
